const WORKER_API_URL = "https://boxmon.boxmon25.workers.dev/api/prices"; async function syncLiveMarketPrices() { try { const res = await fetch(WORKER_API_URL); if (!res.ok) return; const data = await res.json(); if (data && data.prices) { SETS_DATA.forEach(set => { // Check both standard set ID and variant-specific IDs (e.g. sv3pt5-151-etb) const liveInfo = data.prices[set.id] || (set.selectedType === 'ETB' ? data.prices[`${set.id}-etb`] : null); if (liveInfo && set.variants[set.selectedType]) { const hist = set.variants[set.selectedType].history; // Update latest non-null price point across all intervals Object.keys(hist).forEach(scale => { const arr = hist[scale]; for (let i = arr.length - 1; i >= 0; i--) { if (arr[i] !== null && arr[i] !== undefined) { arr[i] = liveInfo.currentMarket; break; } } }); } }); // Re-render UI components with live data renderMarketMovers(); updateMainChart(); renderCatalog(); } } catch (err) { console.warn("Using offline market baseline:", err); } } window.addEventListener('DOMContentLoaded', () => { renderMarketMovers(); updateMainChart(); renderCatalog(); syncLiveMarketPrices(); // Fetches live worker cache });
