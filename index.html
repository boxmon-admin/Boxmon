from PIL import Image # Load original uploaded image img = Image.open("image.png") width, height = img.size # Target boxes defined in [ymin, xmin, ymax, xmax] normalized (0-1000) targets = { "zacian-zamazenta-upc.png": [65, 8, 175, 230], "celebrations-upc.png": [45, 280, 185, 470], "arceus-vstar-upc.png": [60, 510, 170, 735], "swsh-charizard-upc.png": [40, 790, 185, 970], "151-mew-upc.png": [410, 15, 555, 222], "greninja-upc.png": [410, 275, 555, 468], "terapagos-ex-upc.png": [410, 528, 545, 718], "prismatic-evolutions-spc.png": [428, 775, 545, 978], "mega-charizard-x-upc.png": [775, 16, 922, 220], "tr-moltres-upc.png": [798, 275, 898, 465], } for filename, (ymin, xmin, ymax, xmax) in targets.items(): crop_box = ( int(xmin * width / 1000), int(ymin * height / 1000), int(xmax * width / 1000), int(ymax * height / 1000), ) cropped_img = img.crop(crop_box) cropped_img.save(filename) print(f"Exported: {filename} -> {crop_box}")
